    kw = {}; mutate_ev = None; mutate_I = None; rebuild = False; nonce_ok = True

    if plant == "approval-non-delivery": mutate_ev = lambda ev, evs: ev["payload"].update({"before": TF.git(Path(h._wk), "rev-parse", ev["payload"]["head"] + "^"), "head": TF.git(Path(h._wk), "rev-parse", ev["payload"]["head"] + "^"), "commits": []})
    if plant == "pool-digest": mutate_I = lambda I: I.update({"pool_sha256": "q" * 64})
    if plant == "adoption-mismatch": pass   # planted after the build: the adoption file rewritten to a different digest (restored per test)
    if plant == "lock": kw["lock_ok"] = False
    if plant == "nonce-bad": nonce_ok = False
    if plant == "rewrite": rebuild = True
    if plant == "seed-bodies": kw["_seed_bodies"] = True
    env = h.build_plant(rebuild_after=rebuild, mutate_ev=mutate_ev, mutate_I=mutate_I, nonce_ok=nonce_ok, **kw)
    live = env["live"]
    if plant == "open-wrong-repo": env["of"].write_text(json.dumps({**env["evs"][0], "repo": {"name": "wrong/repo"}}))
    if plant == "open-non-delivery": gp = TF.git(env["wk"], "rev-parse", env["evs"][0]["payload"]["before"] + "^"); env["of"].write_text(json.dumps({**env["evs"][0], "payload": {**env["evs"][0]["payload"], "before": gp, "head": gp, "commits": []}}))
    if plant == "same-id-approval": live = [{**env["genuine"], "payload": {**env["genuine"]["payload"], "before": "0" * 40}}] + env["evs"]
    if plant == "same-id-open": live = [env["ev"], {**env["evs"][0], "payload": {**env["evs"][0]["payload"], "before": "0" * 40}}] + env["evs"][1:]
    if plant == "batch": live = [env["ev"], env["evs"][0], {"type": "PushEvent", "id": "batch", "created_at": "2026-09-06T09:59:00Z", "repo": {"name": REPO}, "payload": {"ref": REF, "before": env["evs"][1]["payload"]["head"], "head": env["evs"][-1]["payload"]["head"]}}]

def fail_prereqs(Ps, env):
    ctxs = []; runner = feed(env["live"]); live = env["live"]
    for P_ in Ps:
        if P_ == "identity-file": Path(env["ident"]).unlink()
        elif P_ == "open-file": env["of"].write_text("null")
        elif P_ == "adoption-file": env.setdefault("_adoption_backup", Path(rc.ADOPTION_FILE).read_bytes()); Path(rc.ADOPTION_FILE).unlink()
        elif P_ == "witness-fetch": ctxs.append(mock.patch.object(rc, "verify_witness", side_effect=rc.DataIntegrityFail("WITNESS-FETCH-FAILED fixture transport unavailable")))
        elif P_ == "verifier-import": ctxs.append(mock.patch.object(rc, "_drand_modules", side_effect=rc.DataIntegrityFail("VERIFIER-UNAVAILABLE: fixture — py_ecc unavailable")))
        elif P_ == "helpers-import": ctxs.append(mock.patch.object(rc, "_P", side_effect=ImportError("fixture: provenance helpers unavailable")))
        elif P_ == "events-page1": runner = feed(live, page1_fail=True)
        elif P_ == "events-later-page":
            filler = [{"type": "IssuesEvent", "id": f"f{i}", "created_at": "2026-09-06T09:40:00Z", "repo": {"name": REPO}, "payload": {}} for i in range(100 - len(live))]
            runner = feed(live + filler, page2=lambda cmd: (1, "", "gh: HTTP 503"))
        elif P_ == "remote-head": ctxs.append(mock.patch.object(P, "remote_head", lambda url, ref: None))
        elif P_ == "git-launch":
            real = subprocess.run
            def failing(args, *a, **k):
                if args and args[0] == "git": raise OSError("fixture: git cannot launch")
                return real(args, *a, **k)
            ctxs.append(mock.patch("subprocess.run", side_effect=failing))
        else: raise ValueError(P_)
    return ctxs, runner

def expected(Ps, C):
    """the whole outcome the TABLE predicts: codes per check, blocked checks — independent of the implementation's survivors"""
    exp_codes = {C: {DEFECTS[C]["code"]}}; exp_blocked = set(); pre_reporters = {}
    for P_ in Ps:
        pre = rc.PREREQUISITES[P_]; needers = {k for k, v in rc.INDEPENDENCE.items() if P_ in v["needs"]}
        reporter = {"identity-file": "approval-local", "open-file": "open-local", "git-launch": "history-remote", "witness-fetch": "conjunction", "verifier-import": None, "helpers-import": "helpers", "events-page1": "snapshot", "events-later-page": "snapshot", "remote-head": "history-remote", "adoption-file": "adoption"}[P_]
        if reporter: exp_codes.setdefault(reporter, set()).add(pre["failure_code"])
        for n in needers:
            if n == reporter and P_ not in ("git-launch", "remote-head"): continue
            exp_blocked.add(n)
    return exp_codes, exp_blocked
def _make(Ps, C):
    def test(self):
        self.h._wk = None; env = build_with(self.h, DEFECTS[C]["plant"]); self.h._wk = str(env["wk"]); ctxs, runner = fail_prereqs(Ps, env)
        from contextlib import ExitStack
        try:
            with ExitStack() as st:
                for c in ctxs: st.enter_context(c)
                with self.assertRaises(rc.DataIntegrityFail) as cm: rc.load_identity(env["ident"], rc.Protocol(**{**env["cTP"].__dict__, "events_runner": runner}))
        finally:
            if "_adoption_backup" in env: Path(rc.ADOPTION_FILE).write_bytes(env["_adoption_backup"])                       # the adoption file is shared by the class: restore it (a deleted file cascaded into every later test in the first run 2)
        msg = str(cm.exception); out = rc.LAST_OUTCOME; self.assertIsNotNone(out)
        fs = out["findings"]; exp_codes, exp_blocked = expected(Ps, C)
        self.assertTrue(msg.startswith(DEFECTS[C]["code"]) or DEFECTS[C]["code"] in {f["code"] for f in fs}, f"[{Ps} fail; defect {C}] the defect {DEFECTS[C]['code']} must be contributed at all (behavioural, before attribution): winner {msg[:100]!r}; codes {sorted({f['code'] for f in fs})}")
        by_check = {}
        for f in fs: by_check.setdefault(f["check"], set()).add(f["code"])
        self.assertIn(DEFECTS[C]["code"], by_check.get(C, set()), f"[{Ps} fail; defect {C}] the defect must be contributed by ITS check: got {by_check}")
        if "reason" in DEFECTS[C]: self.assertTrue(any(DEFECTS[C]["reason"] in f["why"] for f in fs if f["check"] == C), f"reason {DEFECTS[C]['reason']}: {[f['why'][:60] for f in fs if f['check'] == C]}")
        for chk, codes_ in exp_codes.items():
            if chk == C: continue
            self.assertTrue(codes_ & by_check.get(chk, set()) or (chk == "helpers" and any(f["code"] == "VERIFIER-UNAVAILABLE" for f in fs)), f"the prerequisite failure {codes_} must be contributed by {chk}: got {by_check}")
        pcodes = {rc.PREREQUISITES[P_]["failure_code"] for P_ in Ps} | {"IDENTITY-SCHEMA", "HISTORY-OPEN-EVENT-MISSING", "ADOPTION-MALFORMED"}
        retry_prereq = any(rc.PREREQUISITES[P_]["class"] == "RETRY-UNAVAILABLE" for P_ in Ps)
        for n in exp_blocked:                                                                            # a check that needs a failed prerequisite: BLOCKED by name, or it CONTRIBUTED (unavailability, the prerequisite's code, or its own refusal reached first)
            ok = n in out["blocked"] or n in by_check or (n == C)
            self.assertTrue(ok, f"check {n} needs {Ps}: it must be BLOCKED by name or have contributed under its own name: blocked={sorted(out['blocked'])}, by_check={by_check}")
        # WHOLE OUTCOME: a check that needs none of the failed prerequisites and carries no planted defect may contribute ONLY (i) the same defect (every check that reads the same input detects it — NSD wants exactly that), (ii) a prerequisite's failure code, or (iii) unavailability when a RETRY-class prerequisite failed
        allowed_codes = {DEFECTS[C]["code"]} | pcodes | ({"RETRY-EVENTS-UNAVAILABLE", "RETRY-HISTORY-CONTINUATION", "RETRY-EVENTS-INCOMPLETE", "STAGE-BLOCKED", "MALFORMED-REMOTE-EVIDENCE"} if retry_prereq or "events-later-page" in Ps else set())
        stray = {}
        for k, v in by_check.items():
            if k in exp_blocked or k in ("helpers", "resolver", C): continue
            extra = {c for c in v if c not in allowed_codes and not (c == "HISTORY-CONTINUATION" and any(("reason" in DEFECTS[C] and DEFECTS[C]["reason"] in f["why"]) or f["class"] == "LOCAL-TERMINAL" for f in fs if f["check"] == k))}
            if extra: stray[k] = extra
        self.assertEqual(stray, {}, f"checks that need none of {Ps} and carry no defect contributed something other than the defect / a prerequisite code / unavailability: {stray}")
        self.assertTrue(all(f.get("why") for f in fs))
        w = PC.resolve([PC.finding(f["class"], f["code"], f["why"], f["stage"], i, f["check"]) for i, f in enumerate(fs)]); self.assertTrue(msg.startswith(w["code"]), f"{msg[:80]!r} vs {w['code']}")
    return test
