def verify_frozen_inputs() -> dict:
    expected = {
        SOURCE_MANIFEST_PATH: SOURCE_MANIFEST_SHA256,
        GLOBAL_ATTEMPT_HISTORY: GLOBAL_ATTEMPT_SHA256,
        PARTITION_ATTEMPT_HISTORY: PARTITION_ATTEMPT_SHA256,
        ORDINARY_GUARD_PATH: ORDINARY_GUARD_SHA256,
        GLOBAL_QUERY_PATH: BASE.GLOBAL_QUERY_SHA256,
    }
    for path, digest in expected.items():
        if not path.exists() or sha_path(path) != digest:
            raise RuntimeError(f"frozen input missing or changed: {path}")
    source = json.loads(SOURCE_MANIFEST_PATH.read_text())
    if len(source.get("entries", [])) != 67 or source.get("coverage") != {"lo": 1, "hi": 662174}:
        raise RuntimeError("source partition manifest shape drift")
    return source

def partial_summary(entries: list[dict], rows: list[dict[str, Decimal | int]]) -> dict:
    combined = combine_rows(rows, require_expected_count=False)
    return {
        "label": "PARTIAL",
        "completed_partitions": len(entries),
        "completed_keyspace_units": sum(entry["key_count"] for entry in entries),
        "completed_ranges": [{"lo": entry["lo"], "hi": entry["hi"]} for entry in entries],
        "coverage_is_full_footprint": False,
        "n_cut6_dered": combined["n_cut6_dered"],
        "sum_cos_theta": str(combined["sum_cos_theta"]),
        "sum_cos2_theta": str(combined["sum_cos2_theta"]),
        "mean_cos_theta": str(combined["mean_cos_theta"]),
        "var_pop_cos_theta": str(combined["var_pop_cos_theta"]),
        "threshold": "0.15",
        "threshold_verdict": None,
    }


def pressure_detected(manifest: dict) -> bool:
    for entry in manifest["entries"]:
        run_dir = Path(entry["run_dir"])
        paths = [run_dir / "service_pressure.json", *run_dir.glob("failed_attempts/*/service_pressure.json")]
        for path in paths:
            if not path.exists():
                continue
            value = json.loads(path.read_text())
            events = value.get("events", [value])
            if any(event.get("signal") in HTTP_PRESSURE_SIGNALS for event in events):
                return True
    return False


setattr(BASE, "pressure_detected", pressure_detected)


def latest_phase(run_dir: Path) -> tuple[str | None, str | None]:
    path = run_dir / "poll_history.json"
    if not path.exists():
        return None, None
    value = json.loads(path.read_text())
    phases = value.get("phases", [])
    phase = phases[-1].get("phase") if phases else None
    return value.get("job_url"), phase


def active_phases(active: dict[int, dict]) -> dict[str, str]:
    values = {}
    for item in active.values():
        job_url, phase = latest_phase(Path(item["entry"]["run_dir"]))
        if job_url and phase:
            values[job_url] = phase
    return values


def queue_stalled(now_monotonic: float, last_progress_monotonic: float, phases: dict[str, str]) -> bool:
    return bool(phases) and all(phase in {"PENDING", "QUEUED"} for phase in phases.values()) and (
        now_monotonic - last_progress_monotonic >= QUEUE_STALL_SECONDS
    )


def lost_attempt_count(entry: dict) -> int:
    return len(list(Path(entry["run_dir"]).glob("failed_attempts/*/failure_record.json")))


def status_value(manifest: dict, active: dict[int, dict], *, stop_reason: str | None = None) -> dict:
    completed = landed(manifest)
    entries = [entry for entry, _ in completed]
    rows = [row for _, row in completed]
    partial = None
    if rows and sum(int(row["n_cut6_dered"]) for row in rows) > 0 and len(entries) < len(manifest["entries"]):
        partial = partial_summary(entries, rows)
    threshold_verdict = None
    if len(entries) == len(manifest["entries"]):
        combined = combine_rows(rows)
        threshold_verdict = "PASS" if combined["threshold_met"] else "FAIL"
    return {
        "updated_utc": now(),
        "result_status": "COMPLETE" if len(entries) == len(manifest["entries"]) else "PARTIAL",
        "completed_partitions": len(entries),
        "partition_count": len(manifest["entries"]),
        "completed_keyspace_units": sum(entry["key_count"] for entry in entries),
        "full_keyspace_units": 662174,
        "partial_running_totals": partial,
        "threshold_verdict": threshold_verdict,
        "active": [value["entry"] for value in active.values()],
        "active_phases": active_phases(active),
        "serial_mode": pressure_detected(manifest),
        "queue_stall_seconds": QUEUE_STALL_SECONDS,
        "deadline_utc": DEADLINE_UTC,
        "deadline_kst": DEADLINE_KST,
        "stop_reason": stop_reason,
    }


def abort_active(active: dict[int, dict], reason: str) -> None:
    for value in list(active.values()):
        BASE.abort_job(value["entry"], reason)
        if value["process"].poll() is None:
            value["process"].terminate()
    for value in list(active.values()):
        try:
            value["process"].wait(timeout=30)
        except subprocess.TimeoutExpired:
            value["process"].kill()
            value["process"].wait(timeout=30)
        value["stdout"].close()
        value["stderr"].close()
    active.clear()


def close_lifecycle(manifest: dict, stop_reason: str) -> None:
    BASE.close_lifecycle(manifest, stop_reason)
    lifecycle = json.loads(LIFECYCLE_PATH.read_text())
    if lifecycle.get("exception_state") != "CLOSED":
        raise RuntimeError("rerun exception lifecycle did not close")
    if lifecycle.get("ordinary_guard_sha256_after") != ORDINARY_GUARD_SHA256:
        raise RuntimeError("ordinary guard hash changed during rerun")
    if lifecycle.get("ordinary_guard_unchanged") is not True:
        raise RuntimeError("ordinary guard before/after hashes differ")
    if lifecycle.get("ordinary_guard_verified_rejects_query_after") is not True:
        raise RuntimeError("ordinary guard rejection was not independently verified after rerun")


def main() -> None:
    manifest = build_manifest()
    LOCK_PATH.parent.mkdir(parents=True, exist_ok=True)
    lock_file = LOCK_PATH.open("a+")
    try:
        fcntl.flock(lock_file.fileno(), fcntl.LOCK_EX | fcntl.LOCK_NB)
    except BlockingIOError as exc:
        raise RuntimeError("partition variance rerun orchestrator already holds the lock") from exc

    BASE.open_lifecycle(manifest)
    active: dict[int, dict] = {}
    stop_reason = "hard_failure"
    last_progress_monotonic = time.monotonic()
    last_progress_utc = now()
    seen_executing_jobs: set[str] = set()
    completed_before = len(landed(manifest))

    def interrupted(signum, frame):
        raise InterruptedError(f"signal_{signum}")

    prior_term = signal.signal(signal.SIGTERM, interrupted)
    prior_int = signal.signal(signal.SIGINT, interrupted)
    try:
        while True:
            completed_count = len(landed(manifest))
            if completed_count == len(manifest["entries"]):
                BASE.write_final(manifest)
                stop_reason = "coverage_complete"
                atomic_json(STATUS_PATH, status_value(manifest, active, stop_reason=stop_reason))
                break
            if datetime.now(timezone.utc) >= parse_utc(DEADLINE_UTC):
                stop_reason = "deadline_reached_1800_kst"
                abort_active(active, stop_reason)
                atomic_json(STATUS_PATH, status_value(manifest, active, stop_reason=stop_reason))
                break

            capacity = 1 if pressure_detected(manifest) else MAX_CONCURRENT
            running_ranges = {(value["entry"]["lo"], value["entry"]["hi"]) for value in active.values()}
            for entry in manifest["entries"]:
                if len(active) >= capacity:
                    break
                if validate_completed(entry) is not None or (entry["lo"], entry["hi"]) in running_ranges:
                    continue
                process, stdout, stderr = BASE.launch(entry)
                active[process.pid] = {"process": process, "entry": entry, "stdout": stdout, "stderr": stderr}
                running_ranges.add((entry["lo"], entry["hi"]))

            failed = None
            service_stop = None
            for pid, value in list(active.items()):
                code = value["process"].poll()
                if code is None:
                    continue
                value["stdout"].close()
                value["stderr"].close()
                del active[pid]
                run_dir = Path(value["entry"]["run_dir"])
                if code != 0:
                    if BASE.recorded_job_is_lost(run_dir):
                        BASE.archive_lost_attempt(run_dir, value["entry"])
                        if lost_attempt_count(value["entry"]) >= MAX_LOST_ATTEMPTS_PER_PARTITION:
                            service_stop = f"service_pressure_lost_attempt_limit_{value['entry']['lo']}_{value['entry']['hi']}"
                            break
                        time.sleep(15)
                    elif (run_dir / "service_pressure.json").exists() or (run_dir / "submission.json").exists():
                        time.sleep(15)
                    else:
                        failed = value["entry"]
                        break
                elif validate_completed(value["entry"]) is None:
                    failed = value["entry"]
                    break
            if service_stop:
                stop_reason = service_stop
                abort_active(active, stop_reason)
                atomic_json(STATUS_PATH, status_value(manifest, active, stop_reason=stop_reason))
                break
            if failed is not None:
                stop_reason = f"hard_failure_{failed['lo']}_{failed['hi']}"
                abort_active(active, stop_reason)
                atomic_json(STATUS_PATH, status_value(manifest, active, stop_reason=stop_reason))
                raise RuntimeError(stop_reason)

            phases = active_phases(active)
            for job_url, phase in phases.items():
                if phase in {"EXECUTING", "COMPLETED"} and job_url not in seen_executing_jobs:
                    seen_executing_jobs.add(job_url)
                    last_progress_monotonic = time.monotonic()
                    last_progress_utc = now()
            completed_count = len(landed(manifest))
            if completed_count > completed_before:
                completed_before = completed_count
                last_progress_monotonic = time.monotonic()
                last_progress_utc = now()

            atomic_json(
                QUEUE_WATCH_PATH,
                {
                    "updated_utc": now(),
                    "last_scheduler_progress_utc": last_progress_utc,
                    "queue_stall_seconds": QUEUE_STALL_SECONDS,
                    "active_phases": phases,
                    "completed_partitions": completed_count,
                    "partial_only_until_full_coverage": True,
                },
            )
            if queue_stalled(time.monotonic(), last_progress_monotonic, phases):
                stop_reason = "queue_stalled_45m"
                abort_active(active, stop_reason)
                atomic_json(STATUS_PATH, status_value(manifest, active, stop_reason=stop_reason))
                break

            atomic_json(STATUS_PATH, status_value(manifest, active))
            time.sleep(5)
    except InterruptedError as exc:
        stop_reason = str(exc)
        abort_active(active, stop_reason)
        atomic_json(STATUS_PATH, status_value(manifest, active, stop_reason=stop_reason))
        raise
    finally:
        if active:
            abort_active(active, stop_reason)
        close_lifecycle(manifest, stop_reason)
        signal.signal(signal.SIGTERM, prior_term)
        signal.signal(signal.SIGINT, prior_int)
        fcntl.flock(lock_file.fileno(), fcntl.LOCK_UN)
        lock_file.close()

    print(json.dumps({"stop_reason": stop_reason, "status": json.loads(STATUS_PATH.read_text())}, sort_keys=True))


if __name__ == "__main__":
    main()
